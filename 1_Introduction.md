### R is a programming language that is commonly used for statistics, data analysis, and graphs.

### A variable is a name that stores a value
# The <- symbol means assign this value to this variable
sample_concentration <- 2.5
# We can also use variables in calculations
dilution_factor <- 10
final_concentration <- sample_concentration / dilution_factor


### R can store different types of data
# Numbers are stored as numeric values
cell_count <- 1500000
temperature <- 37.5
# Text is stored as character data. Text must be inside quotation marks
species <- "Drosophila melanogaster"
treatment <- "control"

### R also has logical values. We can use class() to check the type of a variable
is_control <- TRUE
is_treated <- FALSE
class(is_control)
